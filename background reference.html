// Multi-Celestial Canvas 2D Spotlight Engine
// 4 Celestial Bodies: Diamond Stars, Crescent Moons, Ringed Planets, Spiral Galaxies
const canvas = document.getElementById('starGridCanvas');
const ctx = canvas.getContext('2d');

// Renders celestial objects with quadratic falloff and chromatic bloom
function renderCelestialField(mouse, radius, filterMode) {
  celestialGrid.forEach(obj => {
    if (filterMode !== 'all' && obj.type !== filterMode) return;
    const dist = Math.hypot(obj.x - mouse.x, obj.y - mouse.y);
    const inLight = dist < radius;
    const factor = inLight ? Math.cos((dist / radius) * (Math.PI / 2)) : 0;
    
    // Draw distinct procedural celestial entity
    switch(obj.type) {
      case 'star':   drawChromaticStar(ctx, obj, factor); break;
      case 'moon':   drawGlowingCrescent(ctx, obj, factor); break;
      case 'planet': drawRingedPlanet(ctx, obj, factor); break;
      case 'galaxy': drawSpiralGalaxy(ctx, obj, factor, time); break;
    }
  });
}
